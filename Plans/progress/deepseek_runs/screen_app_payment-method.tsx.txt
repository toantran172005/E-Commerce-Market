import React, { useCallback, useState } from 'react';
import {
  Modal,
  Platform,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  View,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { Ionicons } from '@expo/vector-icons';
import { useRouter } from 'expo-router';

import { colors, radius, spacing } from '../src/constants/theme';
import { usePaymentMethods } from '../src/context/PaymentMethodsContext';
import {
  Chip,
  ConfirmSheet,
  EmptyState,
  FormField,
  LoadingView,
  PrimaryButton,
  ScreenHeader,
  SecondaryButton,
  Toast,
} from '../src/components';
import { PaymentMethod, PaymentMethodType } from '../src/types';

type IoniconName = keyof typeof Ionicons.glyphMap;

function getPaymentIcon(type: PaymentMethodType): IoniconName {
  switch (type) {
    case 'card':
      return 'card-outline';
    case 'wallet':
      return 'wallet-outline';
    case 'cod':
      return 'cash-outline';
    default:
      return 'card-outline';
  }
}

export default function PaymentMethodScreen() {
  const router = useRouter();
  const {
    methods,
    defaultMethod,
    loading,
    addMethod,
    removeMethod,
    setDefault,
  } = usePaymentMethods();

  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [methodToDelete, setMethodToDelete] = useState<PaymentMethod | null>(null);
  const [showAddModal, setShowAddModal] = useState(false);
  const [formType, setFormType] = useState<PaymentMethodType>('card');
  const [formName, setFormName] = useState('');
  const [formDetail, setFormDetail] = useState('');
  const [formExpiry, setFormExpiry] = useState('');
  const [formError, setFormError] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [confirming, setConfirming] = useState(false);
  const [toastVisible, setToastVisible] = useState(false);
  const [toastMessage, setToastMessage] = useState('');

  const selectedMethod =
    methods.find((m) => m.id === selectedId) ?? defaultMethod ?? methods[0] ?? null;

  const showToast = useCallback((message: string) => {
    setToastMessage(message);
    setToastVisible(true);
  }, []);

  const openAddModal = useCallback(() => {
    setFormType('card');
    setFormName('');
    setFormDetail('');
    setFormExpiry('');
    setFormError('');
    setShowAddModal(true);
  }, []);

  const handleAddMethod = useCallback(async () => {
    const name = formName.trim();
    const detail = formDetail.trim();
    const expiry = formExpiry.trim();
    if (!name || !detail || !expiry) {
      setFormError('Vui lòng nhập đầy đủ thông tin.');
      return;
    }
    setSubmitting(true);
    try {
      await addMethod({
        type: formType,
        name,
        detail,
        expiry,
        isDefault: methods.length === 0,
      });
      setShowAddModal(false);
      showToast('Đã thêm phương thức thanh toán');
    } catch (e) {
      setFormError(e instanceof Error ? e.message : 'Có lỗi khi thêm phương thức.');
    } finally {
      setSubmitting(false);
    }
  }, [addMethod, formType, formName, formDetail, formExpiry, methods.length, showToast]);

  const handleDeleteMethod = useCallback(
    (method: PaymentMethod) => setMethodToDelete(method),
    [],
  );

  const confirmDelete = useCallback(async () => {
    if (!methodToDelete) return;
    setSubmitting(true);
    try {
      await removeMethod(methodToDelete.id);
      if (selectedId === methodToDelete.id) {
        setSelectedId(null);
      }
      setMethodToDelete(null);
      showToast('Đã xoá phương thức thanh toán');
    } finally {
      setSubmitting(false);
    }
  }, [methodToDelete, removeMethod, selectedId, showToast]);

  const handleConfirm = useCallback(async () => {
    if (!selectedMethod) {
      showToast('Vui lòng chọn phương thức thanh toán.');
      return;
    }
    setConfirming(true);
    try {
      await setDefault(selectedMethod.id);
      router.back();
    } finally {
      setConfirming(false);
    }
  }, [selectedMethod, setDefault, router, showToast]);

  const selectedIsDefault = selectedMethod && defaultMethod?.id === selectedMethod.id;

  return (
    <SafeAreaView style={styles.safeArea} edges={['top', 'bottom']}>
      <ScreenHeader title="Phương thức thanh toán" />

      {loading ? (
        <LoadingView label="Đang tải phương thức..." />
      ) : methods.length === 0 ? (
        <EmptyState
          icon="card-outline"
          title="Chưa có phương thức"
          subtitle="Thêm thẻ hoặc ví để thanh toán nhanh hơn."
          actionLabel="Thêm ngay"
          onAction={openAddModal}
        />
      ) : (
        <ScrollView
          style={styles.scroll}
          contentContainerStyle={styles.scrollContent}
          showsVerticalScrollIndicator={false}
        >
          <View style={styles.cardMockup}>
            <View style={styles.cardDecorationOne} />
            <View style={styles.cardDecorationTwo} />

            <View style={styles.cardTopRow}>
              <View style={styles.chip}>
                <View style={styles.chipLine} />
                <View style={styles.chipLine} />
              </View>

              <View style={styles.cardBrandRow}>
                {selectedMethod?.type === 'card' ? (
                  <>
                    <Text style={styles.cardBrandText}>Mastercard</Text>
                    <View style={styles.mastercardLogo}>
                      <View style={[styles.logoCircle, styles.logoCircleRed]} />
                      <View style={[styles.logoCircle, styles.logoCircleYellow]} />
                    </View>
                  </>
                ) : selectedMethod?.type === 'wallet' ? (
                  <Ionicons name="wallet-outline" size={22} color="rgba(255,255,255,0.72)" />
                ) : (
                  <Ionicons name="cash-outline" size={22} color="rgba(255,255,255,0.72)" />
                )}
              </View>
            </View>

            <Text style={styles.cardNumber}>
              {selectedMethod?.type === 'card'
                ? selectedMethod.detail
                : selectedMethod?.type === 'wallet'
                  ? selectedMethod.detail
                  : 'Thanh toán khi nhận hàng'}
            </Text>

            <View style={styles.cardBottomRow}>
              <View>
                <Text style={styles.cardLabel}>CHỦ THẺ</Text>
                <Text style={styles.cardValue}>{selectedMethod?.name ?? ''}</Text>
              </View>
              {selectedMethod?.expiry ? (
                <View style={styles.cardExpiryBlock}>
                  <Text style={styles.cardLabel}>HẾT HẠN</Text>
                  <Text style={styles.cardValue}>{selectedMethod.expiry}</Text>
                </View>
              ) : null}
            </View>

            {selectedIsDefault ? (
              <View style={styles.defaultBadge}>
                <Ionicons name="checkmark" size={12} color={colors.primary} />
                <Text style={styles.defaultBadgeText}>MẶC ĐỊNH</Text>
              </View>
            ) : null}
          </View>

          <Text style={styles.sectionTitle}>PHƯƠNG THỨC ĐÃ LƯU</Text>

          {methods.map((method) => {
            const isSelected = selectedMethod?.id === method.id;
            return (
              <View key={method.id} style={[styles.optionRow, isSelected && styles.optionRowActive]}>
                <Pressable
                  style={styles.optionMainPress}
                  onPress={() => setSelectedId(method.id)}
                  accessibilityRole="button"
                  accessibilityLabel={`Chọn ${method.name}`}
                  accessibilityState={{ selected: isSelected }}
                >
                  <View style={[styles.optionIconBox, isSelected && styles.optionIconBoxActive]}>
                    <Ionicons
                      name={getPaymentIcon(method.type)}
                      size={20}
                      color={isSelected ? colors.primary : colors.gray}
                    />
                  </View>

                  <View style={styles.optionTextBlock}>
                    <Text style={[styles.optionTitle, isSelected && styles.optionTitleActive]}>
                      {method.name}
                    </Text>
                    <Text style={styles.optionSubtitle}>
                      {method.detail}{method.expiry ? ` • ${method.expiry}` : ''}
                    </Text>
                  </View>

                  <View style={[styles.radioOuter, isSelected && styles.radioOuterActive]}>
                    {isSelected ? <View style={styles.radioInner} /> : null}
                  </View>
                </Pressable>

                <Pressable
                  onPress={() => handleDeleteMethod(method)}
                  style={({ pressed }) => [styles.deleteButton, pressed && styles.pressed]}
                  hitSlop={8}
                  accessibilityRole="button"
                  accessibilityLabel={`Xoá ${method.name}`}
                >
                  <Ionicons name="trash-outline" size={16} color={colors.primary} />
                </Pressable>
              </View>
            );
          })}

          <Pressable
            onPress={openAddModal}
            style={({ pressed }) => [styles.addNewCard, pressed && styles.pressed]}
            accessibilityRole="button"
            accessibilityLabel="Thêm phương thức thanh toán"
          >
            <Ionicons name="add" size={18} color={colors.primary} />
            <Text style={styles.addNewCardText}>Thêm phương thức mới</Text>
          </Pressable>
        </ScrollView>
      )}

      <View style={styles.footer}>
        <PrimaryButton
          title="Xác nhận"
          onPress={handleConfirm}
          loading={confirming}
          icon="arrow-forward"
          style={styles.confirmButton}
        />
      </View>

      <ConfirmSheet
        visible={methodToDelete !== null}
        title="Xoá phương thức?"
        message={
          methodToDelete
            ? `Bạn có chắc muốn xoá ${methodToDelete.name} (${methodToDelete.detail})?`
            : ''
        }
        confirmText="Xoá"
        cancelText="Huỷ"
        danger
        onConfirm={confirmDelete}
        onCancel={() => setMethodToDelete(null)}
      />

      <Modal
        visible={showAddModal}
        animationType="slide"
        transparent
        onRequestClose={() => setShowAddModal(false)}
      >
        <View style={styles.modalBackdrop}>
          <View style={styles.modalCard}>
            <Text style={styles.modalTitle}>Thêm phương thức thanh toán</Text>

            <View style={styles.typeChips}>
              <Chip
                label="Thẻ"
                selected={formType === 'card'}
                onPress={() => setFormType('card')}
              />
              <Chip
                label="Ví"
                selected={formType === 'wallet'}
                onPress={() => setFormType('wallet')}
              />
              <Chip
                label="COD"
                selected={formType === 'cod'}
                onPress={() => setFormType('cod')}
              />
            </View>

            <FormField
              label="Tên phương thức"
              value={formName}
              onChangeText={setFormName}
              placeholder="Ví dụ: Visa ngân hàng"
            />
            <FormField
              label="Chi tiết"
              value={formDetail}
              onChangeText={setFormDetail}
              placeholder={
                formType === 'card'
                  ? 'Ví dụ: Visa •••• 4242'
                  : formType === 'wallet'
                    ? 'Ví dụ: alex@email'
                    : 'Thanh toán khi nhận hàng'
              }
            />
            <FormField
              label="Hạn / ghi chú"
              value={formExpiry}
              onChangeText={setFormExpiry}
              placeholder="Ví dụ: 09/28"
              keyboardType="numbers-and-punctuation"
            />

            {formError ? <Text style={styles.formError}>{formError}</Text> : null}

            <View style={styles.modalActions}>
              <SecondaryButton
                title="Huỷ"
                onPress={() => setShowAddModal(false)}
                style={styles.modalButton}
              />
              <PrimaryButton
                title="Thêm"
                onPress={handleAddMethod}
                loading={submitting}
                disabled={submitting}
                style={styles.modalButton}
              />
            </View>
          </View>
        </View>
      </Modal>

      <Toast
        visible={toastVisible}
        message={toastMessage}
        onHide={() => setToastVisible(false)}
      />
    </SafeAreaView>
  );
}
