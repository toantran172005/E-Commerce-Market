type PaymentBrandKey = 'momo' | 'vietcombank' | 'bidv' | 'vietinbank' | 'agribank' | 'cod';

function getBrandMeta(method: PaymentMethod | null): { color: string; shortName: string } {
  if (!method) return { color: colors.primary, shortName: 'PAY' };

  if (method.type === 'cod') {
    return { color: method.color ?? '#1E293B', shortName: 'COD' };
  }

  const brand = PAYMENT_BRANDS.find((b) => b.label === method.name);
  const fallbackShortName = method.name.slice(0, 4).toUpperCase();

  return {
    color: method.color ?? brand?.color ?? colors.primary,
    shortName: brand?.shortName ?? fallbackShortName,
  };
}

function getPaymentIcon(type: PaymentMethodType): IoniconName {
  switch (type) {
    case 'card':
      return 'card-outline';
    case 'wallet':
      return 'wallet-outline';
    case 'cod':
      return 'cash-outline' as IoniconName;
    default:
      return 'card-outline';
  }
}

function maskNumber(raw: string): string {
  const last4 = raw.replace(/\s/g, '').slice(-4);
  return last4 ? `•••• ${last4}` : raw;
}

function getCardNumber(method: PaymentMethod | null): string {
  if (!method) return '•••• •••• •••• 1234';
  if (method.type === 'cod') return 'Thanh toán khi nhận hàng';

  const last4 = method.detail.replace(/\s/g, '').replace(/[^0-9]/g, '').slice(-4);
  return last4 ? `•••• •••• •••• ${last4}` : method.detail;
}

export default function PaymentMethodScreen() {
  const router = useRouter();
  const {
    methods,
    defaultMethod,
    loading,
    addMethod,
    removeMethod,
    setDefault,
  } = usePaymentMethods();

  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [methodToDelete, setMethodToDelete] = useState<PaymentMethod | null>(null);
  const [showAddModal, setShowAddModal] = useState(false);
  const [selectedBrandKey, setSelectedBrandKey] = useState<PaymentBrandKey | null>(null);
  const [formDetail, setFormDetail] = useState('');
  const [formExpiry, setFormExpiry] = useState('');
  const [formError, setFormError] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [confirming, setConfirming] = useState(false);
  const [toastVisible, setToastVisible] = useState(false);
  const [toastMessage, setToastMessage] = useState('');

  const selectedMethod =
    methods.find((m) => m.id === selectedId) ?? defaultMethod ?? methods[0] ?? null;

  const showToast = useCallback((message: string) => {
    setToastMessage(message);
    setToastVisible(true);
  }, []);

  const openAddModal = useCallback(() => {
    setSelectedBrandKey(null);
    setFormDetail('');
    setFormExpiry('');
    setFormError('');
    setShowAddModal(true);
  }, []);

  const handleAddMethod = useCallback(async () => {
    if (!selectedBrandKey) {
      setFormError('Vui lòng chọn hình thức thanh toán.');
      return;
    }
    const brand = PAYMENT_BRANDS.find((b) => b.key === selectedBrandKey);
    if (!brand) return;

    const detailRaw = formDetail.trim();
    const expiry = formExpiry.trim();
    if (!detailRaw || !expiry) {
      setFormError('Vui lòng nhập đầy đủ thông tin.');
      return;
    }

    setSubmitting(true);
    try {
      await addMethod({
        type: brand.type,
        name: brand.label,
        detail: `${brand.label} ${maskNumber(detailRaw)}`,
        expiry,
        isDefault: false,
        color: brand.color,
      });
      setShowAddModal(false);
      showToast('Đã thêm phương thức thanh toán');
    } catch (e) {
      setFormError(e instanceof Error ? e.message : 'Có lỗi khi thêm phương thức.');
    } finally {
      setSubmitting(false);
    }
  }, [addMethod, selectedBrandKey, formDetail, formExpiry, showToast]);

  const handleDeleteMethod = useCallback(
    (method: PaymentMethod) => setMethodToDelete(method),
    [],
  );

  const confirmDelete = useCallback(async () => {
    if (!methodToDelete) return;
    setSubmitting(true);
    try {
      await removeMethod(methodToDelete.id);
      if (selectedId === methodToDelete.id) {
        setSelectedId(null);
      }
      setMethodToDelete(null);
      showToast('Đã xoá phương thức thanh toán');
    } finally {
      setSubmitting(false);
    }
  }, [methodToDelete, removeMethod, selectedId, showToast]);

  const handleConfirm = useCallback(async () => {
    if (!selectedMethod) {
      showToast('Vui lòng chọn phương thức thanh toán.');
      return;
    }
    setConfirming(true);
    try {
      await setDefault(selectedMethod.id);
      router.back();
    } finally {
      setConfirming(false);
    }
  }, [selectedMethod, setDefault, router, showToast]);

/** Xác định icon cho nhãn địa chỉ: chỉ dùng briefcase-outline cho văn phòng/công ty, còn lại home-outline. */
function getAddressLabelIcon(label: string): IoniconName {
  const normalized = label.toLowerCase();
  if (
    normalized.includes('văn phòng') ||
    normalized.includes('công ty') ||
    normalized.includes('office')
  ) {
    return 'briefcase-outline';
  }
  return 'home-outline';
}

  isSelected,
  onPress,
  onSetDefault,
  onEdit,
  onDelete,
}: AddressCardProps) {
  return (
    <View style={[styles.card, isSelected && styles.cardSelected]}>
      <Pressable
        onPress={onPress}
        style={styles.cardMainPress}
        accessibilityRole="button"
        accessibilityLabel={`Địa chỉ ${address.label}, ${address.fullName}, ${formatPhone(address.phone)}`}
        accessibilityState={{ selected: isSelected }}
      >
        <View style={styles.cardTop}>
          <View style={styles.typeIcon}>
            <Ionicons
              name={getAddressLabelIcon(address.label)}
              size={19}
              color={colors.primary}
            />
          </View>
          <Text style={styles.typeLabel}>{address.label}</Text>

          {address.isDefault ? (
            <View style={styles.defaultTag}>
              <Text style={styles.defaultTagText}>MẶC ĐỊNH</Text>
            </View>
          ) : null}

          <View style={[styles.radio, isSelected && styles.radioSelected]}>
            {isSelected ? <View style={styles.radioDot} /> : null}
          </View>
        </View>

        <View style={styles.infoRow}>
          <Ionicons name="person-outline" size={15} color={colors.textMuted} style={styles.infoIcon} />
          <Text style={styles.nameText}>{address.fullName}</Text>
        </View>

        <View style={styles.infoRow}>
          <Ionicons name="call-outline" size={15} color={colors.textMuted} style={styles.infoIcon} />
          <Text style={styles.phoneText}>{formatPhone(address.phone)}</Text>
        </View>

        <View style={styles.infoRow}>
          <Ionicons name="location-outline" size={15} color={colors.textMuted} style={styles.infoIcon} />
          <Text style={styles.addressText}>{address.address}</Text>
        </View>
      </Pressable>

      <View style={styles.divider} />

      <View style={styles.cardActions}>
        {!address.isDefault ? (
          <Pressable
            onPress={onSetDefault}
            style={({ pressed }) => [styles.actionBtn, styles.setDefaultBtn, pressed && styles.pressed]}
            hitSlop={8}
            accessibilityRole="button"
            accessibilityLabel="Đặt làm mặc định"
          >
            <Ionicons name="checkmark-circle-outline" size={16} color={colors.primary} />
            <Text style={styles.setDefaultText}>Đặt làm mặc định</Text>
          </Pressable>
        ) : (
          <View style={styles.setDefaultPlaceholder} />
        )}

        <View style={styles.actionsRight}>
          <Pressable
            onPress={onEdit}
            style={({ pressed }) => [styles.actionBtn, pressed && styles.pressed]}
            hitSlop={8}
            accessibilityRole="button"
            accessibilityLabel={`Chỉnh sửa địa chỉ ${address.label}`}
          >
            <Ionicons name="create-outline" size={17} color={colors.textSecondary} />
            <Text style={styles.actionText}>Chỉnh sửa</Text>
          </Pressable>

          <Pressable
            onPress={onDelete}
            style={({ pressed }) => [styles.actionBtn, pressed && styles.pressed]}
            hitSlop={8}
            accessibilityRole="button"
            accessibilityLabel={`Xóa địa chỉ ${address.label}`}
          >
            <Ionicons name="trash-outline" size={17} color={colors.danger} />
            <Text style={[styles.actionText, styles.deleteText]}>Xóa</Text>
          </Pressable>
        </View>
      </View>
    </View>
  );
}

export default function ShippingAddressScreen() {
  const router = useRouter();
  const { user } = useAuth();
  const { addresses, defaultAddress, loading, addAddress, updateAddress, removeAddress, setDefault } =
    useAddresses();
  const { addressId: checkoutAddressId, setAddressId } = useCheckout();

  // Chọn địa chỉ hiện tại từ checkout nếu có, ngược lại dùng mặc định.
  const initialSelectedId = checkoutAddressId || defaultAddress?.id || addresses[0]?.id || '';
  const [selectedId, setSelectedId] = useState(initialSelectedId);

  // Modal thêm / sửa địa chỉ.
  const [modalVisible, setModalVisible] = useState(false);
  const [editingAddress, setEditingAddress] = useState<Address | null>(null);
  const [form, setForm] = useState<AddressForm>({
    label: 'Nhà riêng',
    fullName: user?.fullName ?? '',
    phone: user?.phoneNumber ?? '',
    address: '',
    isDefault: false,
  });
  const [formErrors, setFormErrors] = useState<AddressFormErrors>({});
  const [submitting, setSubmitting] = useState(false);

  // Xác nhận xoá.
  const [addressToDelete, setAddressToDelete] = useState<Address | null>(null);

  // Toast thông báo.
  const [toastVisible, setToastVisible] = useState(false);
  const [toastMessage, setToastMessage] = useState('');

  // Tự đồng bộ selectedId khi danh sách thay đổi.
  useEffect(() => {
    if (!addresses.some((a) => a.id === selectedId)) {
      setSelectedId(defaultAddress?.id ?? addresses[0]?.id ?? '');
    }
  }, [addresses, selectedId, defaultAddress]);

  const handleSelect = (id: string) => setSelectedId(id);

  const handleConfirm = () => {
    if (!selectedId) {
      Alert.alert('Chưa chọn địa chỉ', 'Vui lòng chọn một địa chỉ nhận hàng.');
      return;
    }
    setAddressId(selectedId);
    setToastMessage('Đã cập nhật địa chỉ nhận hàng thành công!');
    setToastVisible(true);
  };

  const openAddModal = () => {
    setEditingAddress(null);
    setForm({
      label: 'Nhà riêng',
      fullName: user?.fullName ?? '',
      phone: user?.phoneNumber ?? '',
      address: '',
      isDefault: false,
    });
    setFormErrors({});
    setModalVisible(true);
  };

  const openEditModal = (address: Address) => {
    setEditingAddress(address);
    setForm({
      label: address.label,
      fullName: address.fullName,
      phone: address.phone,
      address: address.address,
      isDefault: address.isDefault,
    });
    setFormErrors({});
    setModalVisible(true);
  };

  const closeModal = () => setModalVisible(false);

  const setQuickLabel = (label: string) => {
    setForm((prev) => ({ ...prev, label }));
    setFormErrors((prev) => ({ ...prev, label: undefined }));
  };

  const validateForm = (): boolean => {
    const errors: AddressFormErrors = {};

    if (!form.label.trim()) errors.label = 'Vui lòng nhập nhãn địa chỉ';
    if (!form.fullName.trim()) errors.fullName = 'Vui lòng nhập họ tên';
    if (!isValidPhoneVn(form.phone)) errors.phone = 'Số điện thoại không hợp lệ';
    if (!form.address.trim()) errors.address = 'Vui lòng nhập địa chỉ cụ thể';

    setFormErrors(errors);
    return Object.keys(errors).length === 0;
  };

  const handleSubmitForm = async () => {
    if (!validateForm()) return;

    setSubmitting(true);
    try {
      if (editingAddress) {
        await updateAddress(editingAddress.id, form);
      } else {
        await addAddress(form);
      }
      setModalVisible(false);
      setToastMessage('Đã lưu địa chỉ thành công!');
      setToastVisible(true);
    } catch (error) {
      Alert.alert('Lỗi', 'Không thể lưu địa chỉ. Vui lòng thử lại.');
    } finally {
      setSubmitting(false);
    }
  };

  const handleDelete = (address: Address) => {
    setAddressToDelete(address);
  };

  const confirmDelete = async () => {
    if (!addressToDelete) return;
    try {
      await removeAddress(addressToDelete.id);
      setAddressToDelete(null);
    } catch (error) {
      Alert.alert('Lỗi', 'Không thể xoá địa chỉ. Vui lòng thử lại.');
    }
  };

  const handleSetDefault = async (address: Address) => {
    try {
      await setDefault(address.id);
      setToastMessage('Đã đặt địa chỉ mặc định thành công!');
      setToastVisible(true);
    } catch (error) {
      Alert.alert('Lỗi', 'Không thể đặt mặc định. Vui lòng thử lại.');
    }
  };

  const renderAddress = ({ item }: { item: Address }) => (
    <AddressCard
      address={item}
      isSelected={selectedId === item.id}
      onPress={() => handleSelect(item.id)}
      onSetDefault={() => handleSetDefault(item)}
      onEdit={() => openEditModal(item)}
      onDelete={() => handleDelete(item)}
    />
  );

  const listHeader = !loading && addresses.length > 0 ? (
    <Text style={styles.sectionHint}>
      Chọn địa chỉ bạn muốn nhận đơn hàng. Bạn có thể chỉnh sửa hoặc xóa địa chỉ bất cứ lúc nào.
    </Text>
  ) : undefined;

  const listEmpty = loading ? (
    <LoadingView label="Đang tải địa chỉ..." />
  ) : (
    <EmptyState
      icon="location-outline"
      title="Chưa có địa chỉ nào"
      subtitle="Thêm địa chỉ nhận hàng để tiến hành đặt đơn nhanh hơn."
      actionLabel="Thêm địa chỉ mới"
      onAction={openAddModal}
    />
  );

  return (
    <SafeAreaView style={styles.safe} edges={['top', 'bottom']}>
      <ScreenHeader
        title="Địa chỉ nhận hàng"
        right={
          <Pressable
            onPress={openAddModal}
            style={({ pressed }) => [styles.headerAddBtn, pressed && styles.pressed]}
            hitSlop={8}
            accessibilityRole="button"
            accessibilityLabel="Thêm địa chỉ mới"
          >
            <Ionicons name="add" size={22} color={colors.primary} />
            <Text style={styles.headerAddText}>Thêm</Text>
          </Pressable>
        }
      />

      <FlatList
        data={addresses}
        keyExtractor={(item) => item.id}
        renderItem={renderAddress}
        ListHeaderComponent={listHeader}
        ListEmptyComponent={listEmpty}
        ListFooterComponent={
          !loading && addresses.length > 0 ? (
            <Pressable
              onPress={openAddModal}
              style={({ pressed }) => [styles.addNewCard, pressed && styles.pressed]}
              accessibilityRole="button"
              accessibilityLabel="Thêm địa chỉ nhận hàng mới"
            >
              <Ionicons name="add-circle-outline" size={26} color={colors.primary} />
              <Text style={styles.addNewCardText}>Thêm địa chỉ nhận hàng mới</Text>
            </Pressable>
          ) : undefined
        }
        contentContainerStyle={styles.listContent}
        showsVerticalScrollIndicator={false}
      />

      <View style={styles.footer}>
        <PrimaryButton
          title="Xác nhận địa chỉ"
          onPress={handleConfirm}
          disabled={!selectedId}
        />
      </View>

      {/* Modal thêm / sửa địa chỉ */}
      <Modal
        visible={modalVisible}
        animationType="slide"
        presentationStyle="pageSheet"
        onRequestClose={closeModal}
      >
        <SafeAreaView style={styles.modalSafe} edges={['top', 'bottom']}>
          <ScreenHeader
            title={editingAddress ? 'Chỉnh sửa địa chỉ' : 'Thêm địa chỉ mới'}
            onBack={closeModal}
          />
          <View style={styles.modalContent}>
            <FormField
              label="Nhãn địa chỉ"
              value={form.label}
              onChangeText={(text) => setForm((prev) => ({ ...prev, label: text }))}
              placeholder="Ví dụ: Nhà riêng, Văn phòng"
              error={formErrors.label}
              icon="pricetag-outline"
            />

            {/* Chip chọn nhanh */}
            <View style={styles.labelChipRow}>
              <Pressable
                onPress={() => setQuickLabel('Nhà riêng')}
                style={[
                  styles.labelChip,
                  form.label === 'Nhà riêng' && styles.labelChipActive,
                ]}
              >
                <Ionicons
                  name="home-outline"
                  size={16}
                  color={form.label === 'Nhà riêng' ? colors.white : colors.primary}
                />
                <Text
                  style={[
                    styles.labelChipText,
                    form.label === 'Nhà riêng' && styles.labelChipTextActive,
                  ]}
                >
                  Nhà riêng
                </Text>
              </Pressable>

              <Pressable
                onPress={() => setQuickLabel('Văn phòng')}
                style={[
                  styles.labelChip,
                  form.label === 'Văn phòng' && styles.labelChipActive,
                ]}
              >
                <Ionicons
                  name="briefcase-outline"
                  size={16}
                  color={form.label === 'Văn phòng' ? colors.white : colors.primary}
                />
                <Text
                  style={[
                    styles.labelChipText,
                    form.label === 'Văn phòng' && styles.labelChipTextActive,
                  ]}
                >
                  Văn phòng
                </Text>
              </Pressable>
            </View>

            <FormField
              label="Họ và tên"
              value={form.fullName}
              onChangeText={(text) => setForm((prev) => ({ ...prev, fullName: text }))}
              placeholder="Nhập họ tên"
              error={formErrors.fullName}
              icon="person-outline"
            />
            <FormField
              label="Số điện thoại"
              value={form.phone}
              onChangeText={(text) => setForm((prev) => ({ ...prev, phone: text }))}
              placeholder="0901234567"
              error={formErrors.phone}
              icon="call-outline"
              keyboardType="phone-pad"
            />
            <FormField
              label="Địa chỉ"
              value={form.address}
              onChangeText={(text) => setForm((prev) => ({ ...prev, address: text }))}
              placeholder="Số nhà, đường, quận, thành phố"
              error={formErrors.address}
              icon="location-outline"
              multiline
            />

            <PrimaryButton
              title={editingAddress ? 'Cập nhật địa chỉ' : 'Thêm địa chỉ'}
              onPress={handleSubmitForm}
              loading={submitting}
            />
          </View>
        </SafeAreaView>
      </Modal>

      {/* Xác nhận xoá */}
      <ConfirmSheet
        visible={addressToDelete !== null}
        title="Xóa địa chỉ"
        message={`Bạn có chắc chắn muốn xóa địa chỉ "${addressToDelete?.label}" không?`}
        confirmText="Xóa"
        cancelText="Hủy"
        danger
        onConfirm={confirmDelete}
        onCancel={() => setAddressToDelete(null)}
      />

      {/* Toast thông báo */}
      <Toast
        message={toastMessage}
        visible={toastVisible}
        duration={1200}
        onHide={() => {
          setToastVisible(false);
          // Không tự động router.back() để cho phép chỉnh sửa/xoá liên tục.
          // router.back() vẫn có thể dùng khi cần, nhưng hiện tại không bắt buộc.
        }}
      />
    </SafeAreaView>
  );
}
