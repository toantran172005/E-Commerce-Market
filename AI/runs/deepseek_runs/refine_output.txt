const { width } = Dimensions.get('window');
const H_PADDING = spacing.xl; // 20
const CARD_GAP = spacing.md;   // 12
const CARD_WIDTH = (width - H_PADDING * 2 - CARD_GAP) / 2;

interface HomeProductCardProps {
  product: Product;
  width?: number;
  isFavourite: boolean;
  onToggleFavourite: () => void;
  onPress: () => void;
}

function HomeProductCard({
  product,
  width,
  isFavourite,
  onToggleFavourite,
  onPress,
}: HomeProductCardProps) {
  const discountPercent = calcDiscountPercent(product.price, product.originalPrice);

  return (
    <Pressable
      onPress={onPress}
      style={({ pressed }) => [styles.card, width ? { width } : null, pressed && styles.pressedCard]}
      accessibilityRole="button"
      accessibilityLabel={`${product.name}, giá ${formatMoney(product.price)}`}
    >
      <View style={styles.cardImageWrap}>
        <Image source={{ uri: product.imageUrl }} style={styles.cardImage} resizeMode="contain" />

        {discountPercent != null ? (
          <View style={styles.discountBadge}>
            <Text style={styles.discountText}>-{discountPercent}%</Text>
          </View>
        ) : product.badge === 'new' ? (
          <View style={[styles.discountBadge, styles.newBadge]}>
            <Text style={styles.discountText}>NEW</Text>
          </View>
        ) : null}

        <Pressable
          onPress={onToggleFavourite}
          hitSlop={8}
          style={styles.heartButton}
          accessibilityRole="button"
          accessibilityLabel={isFavourite ? 'Bỏ yêu thích' : 'Thêm yêu thích'}
        >
          <Ionicons
            name={isFavourite ? 'heart' : 'heart-outline'}
            size={16}
            color={isFavourite ? colors.primary : colors.gray}
          />
        </Pressable>
      </View>

      <Text style={styles.cardName} numberOfLines={2}>{product.name}</Text>

      <View style={styles.ratingRow}>
        <RatingStars rating={product.rating ?? 0} size={11} />
        <Text style={styles.reviewCount}>({product.reviewCount ?? 0})</Text>
      </View>

      <View style={styles.priceRow}>
        <Text style={styles.price}>{formatMoney(product.price)}</Text>
        {product.originalPrice && product.originalPrice > product.price ? (
          <Text style={styles.oldPrice}>{formatMoney(product.originalPrice)}</Text>
        ) : null}
      </View>
    </Pressable>
  );
}

      setCategoryIds((prev) => (prev.includes(categoryId) ? [] : [categoryId]));

/** Đổi createdAt ISO thành chuỗi thời gian tương đối tiếng Việt. */
function formatRelativeTime(value: string | number): string {
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return '';

  const diffMinutes = Math.round((Date.now() - date.getTime()) / 60000);
  if (diffMinutes < 1) return 'Vừa xong';
  if (diffMinutes < 60) return `${diffMinutes} phút trước`;

  const diffHours = Math.floor(diffMinutes / 60);
  if (diffHours < 24) return `${diffHours} giờ trước`;

  const diffDays = Math.floor(diffHours / 24);
  if (diffDays < 7) return `${diffDays} ngày trước`;

  return formatDate(value);
}

export default function NotificationScreen() {
  const router = useRouter();
  const insets = useSafeAreaInsets();
  const {
    notifications,
    unreadCount,
    loading,
    markAsRead,
    markAllAsRead,
    loadNotifications,
  } = useNotifications();

  const [refreshing, setRefreshing] = useState(false);
  const [selectedNotification, setSelectedNotification] = useState<AppNotification | null>(null);

  const handleGoBack = useCallback(() => {
    if (router.canGoBack()) {
      router.back();
    } else {
      router.replace('/(tabs)/home');
    }
  }, [router]);

  const handleMarkAllAsRead = useCallback(() => {
    if (unreadCount === 0) return;
    markAllAsRead().catch(() => undefined);
  }, [unreadCount, markAllAsRead]);

  const handlePressNotification = useCallback(
    (item: AppNotification) => {
      // Đánh dấu đã đọc, sau đó mở modal chi tiết.
      if (!item.read) {
        markAsRead(item.id).catch(() => undefined);
      }
      setSelectedNotification(item);
    },
    [markAsRead],
  );

  const handleRefresh = useCallback(async () => {
    setRefreshing(true);
    try {
      await loadNotifications();
    } finally {
      setRefreshing(false);
    }
  }, [loadNotifications]);

  const renderItem: ListRenderItem<AppNotification> = useCallback(
    ({ item }) => {

      return (
        <Pressable

              {item.message}
            </Text>

              <Text style={styles.timeText}>{formatRelativeTime(item.createdAt)}</Text>
            </View>
          </View>
        </Pressable>
      );
    },
    [handlePressNotification],
  );

  const markAllButton = (

          onRefresh={handleRefresh}
        />

    case 'BIDV':
      return { color: '#003C71', shortName: 'BIDV' };
    case 'VietinBank':
      return { color: '#F7941D', shortName: 'VietinBank' };
    case 'Agribank':
      return { color: '#009B74', shortName: 'Agribank' };
    default:
      return { color: colors.primary, shortName: 'COD' };
  }
}

function getPaymentIcon(type: PaymentMethodType): IoniconName {
  switch (type) {
    case 'card':
      return 'card-outline';
    case 'wallet':
      return 'wallet-outline';
    case 'cod':
      return 'cash-outline';
    default:
      return 'card-outline';
  }
}

function maskNumber(raw: string): string {
  const last4 = raw.replace(/\s/g, '').slice(-4);
  return last4 ? `•••• ${last4}` : raw;
}

export default function PaymentMethodScreen() {
  const router = useRouter();
  const {
    methods,
    defaultMethod,
    loading,
    addMethod,
    removeMethod,
    setDefault,
  } = usePaymentMethods();

  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [methodToDelete, setMethodToDelete] = useState<PaymentMethod | null>(null);
  const [showAddModal, setShowAddModal] = useState(false);
  const [selectedBrandKey, setSelectedBrandKey] = useState<PaymentBrandKey | null>(null);
  const [formDetail, setFormDetail] = useState('');
  const [formExpiry, setFormExpiry] = useState('');
  const [formError, setFormError] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [confirming, setConfirming] = useState(false);
  const [toastVisible, setToastVisible] = useState(false);
  const [toastMessage, setToastMessage] = useState('');

  const selectedMethod =
    methods.find((m) => m.id === selectedId) ?? defaultMethod ?? methods[0] ?? null;

  const showToast = useCallback((message: string) => {
    setToastMessage(message);
    setToastVisible(true);
  }, []);

  const openAddModal = useCallback(() => {
    setSelectedBrandKey(null);
    setFormDetail('');
    setFormExpiry('');
    setFormError('');
    setShowAddModal(true);
  }, []);

  const handleAddMethod = useCallback(async () => {
    if (!selectedBrandKey) {
      setFormError('Vui lòng chọn hình thức thanh toán.');
      return;
    }
    const brand = PAYMENT_BRANDS.find((b) => b.key === selectedBrandKey);
    if (!brand) return;

    const detailRaw = formDetail.trim();
    const expiry = formExpiry.trim();
    if (!detailRaw || !expiry) {
      setFormError('Vui lòng nhập đầy đủ thông tin.');
      return;
    }

    setSubmitting(true);
    try {
      await addMethod({
        type: brand.type,
        name: brand.label,
        detail: `${brand.label} ${maskNumber(detailRaw)}`,
        expiry,
        isDefault: false,
      });
      setShowAddModal(false);
      showToast('Đã thêm phương thức thanh toán');
    } catch (e) {
      setFormError(e instanceof Error ? e.message : 'Có lỗi khi thêm phương thức.');
    } finally {
      setSubmitting(false);
    }
  }, [addMethod, selectedBrandKey, formDetail, formExpiry, showToast]);

  const handleDeleteMethod = useCallback(
    (method: PaymentMethod) => setMethodToDelete(method),
    [],
  );

  const confirmDelete = useCallback(async () => {
    if (!methodToDelete) return;
    setSubmitting(true);
    try {
      await removeMethod(methodToDelete.id);
      if (selectedId === methodToDelete.id) {
        setSelectedId(null);
      }
      setMethodToDelete(null);
      showToast('Đã xoá phương thức thanh toán');
    } finally {
      setSubmitting(false);
    }

      showToast('Vui lòng chọn phương thức thanh toán.');
      return;

    setConfirming(true);

      await setDefault(selectedMethod.id);
      router.back();

      setConfirming(false);

function SettingRow({

  title,
  subtitle,
  titleColor = colors.dark,
  right,
  onPress,
  isLast,

      <Pressable

        style={({ pressed }) => [
          styles.row,
          !isLast && styles.rowDivider,
          pressed && { opacity: 0.7 },

        accessibilityLabel={title}
      >
        {content}
      </Pressable>
    );
  }

  return <View style={[styles.row, !isLast && styles.rowDivider]}>{content}</View>;
}

    try {
      await saveSettings(draft);
      Alert.alert('Đã lưu cài đặt', 'Các tuỳ chỉnh của bạn đã được cập nhật.', [
        { text: 'OK', onPress: () => router.back() },
      ]);
    } catch (error) {
      Alert.alert('Lỗi', 'Không thể lưu cài đặt. Vui lòng thử lại.');
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleToggle = (key: keyof AppSettings, value: boolean) => {
    setDraft((prev) => ({ ...prev, [key]: value }));
  };

  const handleBiometricToggle = (value: boolean) => {
    setDraft((prev) => ({ ...prev, biometricEnabled: value }));
    if (value) {
      Alert.alert(
        'Đã bật đăng nhập sinh trắc học',
        'Bạn có thể đăng nhập bằng FaceID / vân tay trên thiết bị này.',
      );
    }
  };

  const handlePrivacyPolicy = () => {
    Alert.alert(
      'Chính sách bảo mật',
      'Chúng tôi chỉ lưu trữ thông tin cần thiết để xử lý đơn hàng. Dữ liệu của bạn không bao giờ được bán cho bên thứ ba và bạn có thể yêu cầu xoá bất cứ lúc nào.',
      [{ text: 'Đã hiểu' }],
    );
  };

  const closePasswordModal = () => {
    setPasswordModalVisible(false);
    setCurrentPassword('');
    setNewPassword('');
    setConfirmPassword('');
    setPasswordError(null);
    setShowPassword(false);
  };

  const handlePasswordSubmit = async () => {
    if (!currentPassword || !newPassword || !confirmPassword) {
      setPasswordError('Vui lòng nhập đầy đủ các trường.');
      return;
    }
    if (newPassword.length < 6) {
      setPasswordError('Mật khẩu mới phải có ít nhất 6 ký tự.');
      return;
    }
    if (newPassword !== confirmPassword) {
      setPasswordError('Mật khẩu xác nhận không khớp.');
      return;
    }
    setIsPasswordSubmitting(true);
    setPasswordError(null);
    try {
      await new Promise((resolve) => setTimeout(resolve, 600));
      Alert.alert('Đổi mật khẩu thành công', 'Mật khẩu của bạn đã được cập nhật.');
      closePasswordModal();
    } catch {
      Alert.alert('Lỗi', 'Không thể đổi mật khẩu. Vui lòng thử lại.');
    } finally {
      setIsPasswordSubmitting(false);
    }
  };

  if (loading) {
    return <LoadingView label="Đang tải cài đặt..." />;
  }

  return (
    <SafeAreaView style={styles.safe} edges={['top', 'bottom']}>
      <View style={styles.header}>
        <Pressable
          onPress={handleBack}
          style={({ pressed }) => [styles.headerButton, pressed && { opacity: 0.7 }]}
          accessibilityRole="button"
          accessibilityLabel="Quay lại"
        >
          <Ionicons name="chevron-back" size={22} color={colors.dark} />
        </Pressable>
        <Text style={styles.headerTitle}>Cài đặt ứng dụng</Text>
        <Pressable
          onPress={handleClose}
          style={({ pressed }) => [styles.headerButton, pressed && { opacity: 0.7 }]}
          accessibilityRole="button"
          accessibilityLabel="Đóng"
        >
          <Ionicons name="close" size={20} color={colors.dark} />
        </Pressable>
      </View>

      <ScrollView
        contentContainerStyle={styles.scrollContent}
        showsVerticalScrollIndicator={false}
        keyboardShouldPersistTaps="handled"
      >
        <SectionLabel>THÔNG BÁO</SectionLabel>
        <View style={styles.card}>
          <SettingRow
            icon="notifications-outline"
            title="Cho phép thông báo"
            subtitle="Nhận thông báo từ ứng dụng"
            isLast
            right={
              <Switch
                value={draft.pushEnabled}
                onValueChange={(value) => handleToggle('pushEnabled', value)}
                trackColor={{ false: colors.border, true: colors.primary }}
                thumbColor={colors.white}
                ios_backgroundColor={colors.border}
                accessibilityLabel="Cho phép thông báo"
                accessibilityRole="switch"
                accessibilityState={{ checked: draft.pushEnabled }}
              />
            }
          />
        </View>

        <SectionLabel>BẢO MẬT</SectionLabel>
        <View style={styles.card}>
          <SettingRow
            icon="lock-closed-outline"
            title="Đổi mật khẩu"
            subtitle="Cập nhật mật khẩu tài khoản"
            onPress={() => setPasswordModalVisible(true)}
            right={<Ionicons name="chevron-forward" size={18} color={colors.textMuted} />}
          />
          <SettingRow
            icon="finger-print-outline"
            title="Đăng nhập sinh trắc học"
            subtitle="FaceID / vân tay"
            isLast
            right={
              <Switch
                value={draft.biometricEnabled}
                onValueChange={handleBiometricToggle}
                trackColor={{ false: colors.border, true: colors.primary }}
                thumbColor={colors.white}
                ios_backgroundColor={colors.border}
                accessibilityLabel="Đăng nhập sinh trắc học"
                accessibilityRole="switch"
                accessibilityState={{ checked: draft.biometricEnabled }}
              />
            }
          />
        </View>

        <SectionLabel>QUYỀN RIÊNG TƯ</SectionLabel>
        <View style={styles.card}>
          <SettingRow
            icon="shield-checkmark-outline"
            title="Chính sách bảo mật"
            subtitle="Cách chúng tôi xử lý dữ liệu"
            onPress={handlePrivacyPolicy}
            right={<Ionicons name="chevron-forward" size={18} color={colors.textMuted} />}
            isLast
          />
        </View>
      </ScrollView>

      <View style={styles.footer}>
        <PrimaryButton
          title="Lưu cài đặt"
          onPress={handleSave}
          loading={isSubmitting}
          disabled={isSubmitting}
        />
      </View>

      <SheetModal
        visible={passwordModalVisible}
        title="Đổi mật khẩu"
        onClose={closePasswordModal}
      >
        <Text style={styles.inputLabel}>Mật khẩu hiện tại</Text>
        <View style={styles.inputWrap}>
          <TextInput
            style={styles.input}
            value={currentPassword}
            onChangeText={setCurrentPassword}
            placeholder="Nhập mật khẩu hiện tại"
            placeholderTextColor={colors.placeholder}
            secureTextEntry={!showPassword}
            autoCapitalize="none"
            autoCorrect={false}
          />
        </View>

        <Text style={styles.inputLabel}>Mật khẩu mới</Text>
        <View style={styles.inputWrap}>
          <TextInput
            style={[styles.input, styles.inputWithIcon]}
            value={newPassword}
            onChangeText={setNewPassword}
            placeholder="Ít nhất 6 ký tự"
            placeholderTextColor={colors.placeholder}
            secureTextEntry={!showPassword}
            autoCapitalize="none"
            autoCorrect={false}
          />
          <Pressable
            style={styles.eyeButton}
            onPress={() => setShowPassword((prev) => !prev)}
            hitSlop={8}
            accessibilityRole="button"
            accessibilityLabel={showPassword ? 'Ẩn mật khẩu' : 'Hiện mật khẩu'}
          >
            <Ionicons
              name={showPassword ? 'eye-off-outline' : 'eye-outline'}
              size={18}
              color={colors.textMuted}
            />
          </Pressable>
        </View>

        <Text style={styles.inputLabel}>Xác nhận mật khẩu mới</Text>
        <View style={styles.inputWrap}>
          <TextInput
            style={styles.input}
            value={confirmPassword}
            onChangeText={setConfirmPassword}
            placeholder="Nhập lại mật khẩu mới"
            placeholderTextColor={colors.placeholder}
            secureTextEntry={!showPassword}
            autoCapitalize="none"
            autoCorrect={false}
          />
        </View>

        {passwordError ? <Text style={styles.errorText}>{passwordError}</Text> : null}

        <View style={styles.sheetActions}>
          <SecondaryButton title="Hủy" onPress={closePasswordModal} style={styles.cancelButton} />
          <PrimaryButton
            title="Cập nhật mật khẩu"
            onPress={handlePasswordSubmit}
            loading={isPasswordSubmitting}
            disabled={isPasswordSubmitting}
            style={styles.updateButton}
          />
        </View>
      </SheetModal>
    </SafeAreaView>
  );
}
