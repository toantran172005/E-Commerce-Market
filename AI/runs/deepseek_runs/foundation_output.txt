/* Các hàm định dạng + validate dùng chung (thuần TypeScript). */

/** Định dạng tiền tệ, mặc định hiển thị $. */
export function formatMoney(value: number, currency: 'USD' | 'VND' = 'USD'): string {
  if (currency === 'VND') {
    return `${Math.round(value).toLocaleString('vi-VN')} ₫`;
  }
  return `$${value.toLocaleString('en-US', {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  })}`;
}

export function formatDate(value: string | number): string {
  const d = new Date(value);
  if (Number.isNaN(d.getTime())) return '';
  const day = d.getDate().toString().padStart(2, '0');
  const month = (d.getMonth() + 1).toString().padStart(2, '0');
  return `${day}/${month}/${d.getFullYear()}`;
}

export function formatDateTime(value: string | number): string {
  const d = new Date(value);
  if (Number.isNaN(d.getTime())) return '';
  const hour = d.getHours().toString().padStart(2, '0');
  const minute = d.getMinutes().toString().padStart(2, '0');
  return `${formatDate(value)} • ${hour}:${minute}`;
}

/** Chuẩn hoá số điện thoại Việt Nam về dạng +84xxx. */
export function formatPhone(phone: string): string {
  const digits = phone.replace(/[^0-9]/g, '');
  if (!digits) return phone;
  if (digits.startsWith('84')) {
    return `+84 ${digits.slice(2, 5)} ${digits.slice(5, 8)} ${digits.slice(8)}`;
  }
  if (digits.startsWith('0')) {
    return `+84 ${digits.slice(1, 4)} ${digits.slice(4, 7)} ${digits.slice(7)}`;
  }
  return phone;
}

export function isValidEmail(email: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim());
}

/** Mật khẩu demo: ít nhất 6 ký tự. */
export function isValidPassword(password: string): boolean {
  return password.length >= 6;
}

/** Số điện thoại VN: bắt đầu 0[35789] hoặc +84, đủ 9-10 chữ số. */
export function isValidPhoneVn(phone: string): boolean {
  const digits = phone.replace(/[^0-9]/g, '');
  if (digits.startsWith('84')) {
    return /^84[35789][0-9]{8}$/.test(digits);
  }
  if (digits.startsWith('0')) {
    return /^0[35789][0-9]{8}$/.test(digits);
  }
  return false;
}

/** Phần trăm giảm giá, trả về số nguyên đã làm tròn. */
export function calcDiscountPercent(price: number, originalPrice?: number): number | null {
  if (!originalPrice || originalPrice <= price) return null;
  return Math.round(((originalPrice - price) / originalPrice) * 100);
}

/** Cắt chuỗi dài, dùng cho tên hiển thị. */
export function truncate(text: string, max = 30): string {
  return text.length <= max ? text : `${text.slice(0, max - 3)}...`;
}

import { CartItem, ShippingMethod, Voucher } from '../types';

export const TAX_RATE = 0.08;

/** Tổng tiền của các sản phẩm ĐƯỢC CHỌN trong giỏ. */
export function computeSubtotal(items: CartItem[]): number {
  return items
    .filter((item) => item.selected)
    .reduce((sum, item) => sum + item.product.price * item.quantity, 0);
}

/** Số tiền giảm từ voucher. Voucher freeShip giảm đúng phí ship. */
export function computeVoucherDiscount(
  voucher: Voucher | null,
  subtotal: number,
  shippingFee: number,
): number {
  if (!voucher || subtotal < voucher.minOrderValue) return 0;
  if (voucher.type === 'percent') {
    const raw = (subtotal * voucher.value) / 100;
    return voucher.maxDiscount ? Math.min(raw, voucher.maxDiscount) : raw;
  }
  if (voucher.type === 'fixed') {
    return Math.min(voucher.value, subtotal);
  }
  return shippingFee;
}

export interface CartTotals {
  subtotal: number;
  discount: number;
  shippingFee: number;
  tax: number;
  total: number;
}

/** Tính đầy đủ: tạm tính, giảm giá, phí ship, thuế, tổng cộng. */
export function computeCartTotals(
  items: CartItem[],
  voucher: Voucher | null,
  shippingMethod: ShippingMethod,
): CartTotals {
  const subtotal = computeSubtotal(items);
  const baseShipping =
    subtotal >= shippingMethod.freeOver ? 0 : shippingMethod.fee;

  const discount = computeVoucherDiscount(voucher, subtotal, baseShipping);
  const isFreeShip = voucher?.type === 'freeShip' && subtotal >= voucher.minOrderValue;
  const shippingFee = isFreeShip ? 0 : baseShipping;

  const taxable = Math.max(subtotal - discount + shippingFee, 0);
  const tax = taxable * TAX_RATE;
  const total = Math.max(subtotal - discount + shippingFee + tax, 0);

  return { subtotal, discount, shippingFee, tax, total };
}

import { CartItem, Product, ProductVariant } from '../types';

/** Tạo CartItem từ Product + biến thể (nếu chọn). */
export function toCartItem(
  product: Product,
  quantity: number,
  variant?: ProductVariant,
): CartItem {
  return {
    id: cartItemKey(product.id, variant?.id),
    product,
    quantity,
    selected: true,
    variantId: variant?.id,
    variantName: variant?.name,
  };
}

/** Key ổn định cho một dòng trong giỏ: sản phẩm + biến thể. */
export function cartItemKey(productId: string, variantId?: string): string {
  return `${productId}::${variantId ?? 'default'}`;
}

/** Tìm CartItem theo key. */
export function findCartItem(items: CartItem[], key: string): CartItem | undefined {
  return items.find((item) => item.id === key || cartItemKey(item.product.id, item.variantId) === key);
}

function toProduct(input: ProductInput, index: number): Product {
  const brand = BRANDS.find((b) => b.id === input.brandId);
  const price = input.price;
  const originalPrice = input.originalPrice ?? input.price;

  return {
    id: input.id,
    name: input.name,
    price,
    originalPrice,
    description: input.description,
    imageUrl: input.images?.[0] ?? pic(input.id),
    images: input.images ?? [pic(input.id)],
    category: input.categoryId,
    rating: input.rating,
    reviewCount: input.reviewCount,
    stock: input.stock,
    soldCount: input.soldCount ?? 0,
    brandId: input.brandId,
    brandName: brand?.name,
    specs: input.specs,
    tags: input.tags ?? [],
    badge: input.badge,
    variants: input.variants ?? [],
    createdAt: input.badge === 'new'
      ? `2026-11-${String(((index * 5) % 28) + 1).padStart(2, '0')}T09:00:00Z`
      : `2026-01-${String(((index * 7) % 28) + 1).padStart(2, '0')}T09:00:00Z`,
  };
}

const addDays = (iso: string, days: number): string => {
  const d = new Date(iso);
  d.setDate(d.getDate() + days);
  return d.toISOString();
};

const fmtDate = (iso: string): string => {
  const d = new Date(iso);
  const day = d.getDate().toString().padStart(2, '0');
  const month = (d.getMonth() + 1).toString().padStart(2, '0');
  return `${day}/${month}/${d.getFullYear()}`;
};

/** Timeline 4 bước, số bước hoàn thành theo trạng thái đơn. */
function buildTimeline(status: OrderStatus, createdAt: string): OrderTimelineStep[] {
  if (status === 'cancelled') {
    return [
      { key: 'ordered', label: 'Đặt hàng', icon: 'cart', date: fmtDate(createdAt), completed: true },
      { key: 'cancelled', label: 'Đã huỷ', icon: 'close', date: fmtDate(addDays(createdAt, 1)), completed: true },
    ];
  }

  const base = [
    { key: 'ordered', label: 'Đặt hàng', icon: 'cart' as const },
    { key: 'processing', label: 'Xử lý', icon: 'cube' as const },
    { key: 'shipped', label: 'Đang giao', icon: 'car' as const },
    { key: 'delivered', label: 'Đã giao', icon: 'home' as const },
  ];

  const completedCount =
    status === 'pending' ? 1 : status === 'processing' ? 2 : status === 'shipped' ? 3 : 4;

  return base.map((step, index) => ({
    key: step.key,
    label: step.label,
    icon: step.icon,
    date: fmtDate(addDays(createdAt, index + (status === 'pending' ? 0 : 1))),
    completed: index < completedCount,
  }));
}

function buildOrder(seed: OrderSeed): Order {
  const address = MOCK_ADDRESSES.find((a) => a.id === seed.addressId) ?? MOCK_ADDRESSES[0];
  const payment = MOCK_PAYMENT_METHODS.find((p) => p.id === seed.paymentMethodId) ?? MOCK_PAYMENT_METHODS[0];
  const shipping = MOCK_SHIPPING_METHODS.find((s) => s.id === seed.shippingMethodId) ?? MOCK_SHIPPING_METHODS[0];
  const voucher = MOCK_VOUCHERS.find((v) => v.code === seed.voucherCode) ?? null;

  const items: OrderItem[] = seed.items.map(({ productId, variantId, quantity }) => {
    const product = MOCK_PRODUCTS.find((p) => p.id === productId);
    return {
      productId,
      name: product?.name ?? 'Sản phẩm đã ẩn',
      price: product?.price ?? 0,
      quantity,
      imageUrl: product?.imageUrl ?? '',
      variantId,
    };
  });

  const subtotal = items.reduce((sum, item) => sum + item.price * item.quantity, 0);
  const baseShipping = subtotal >= shipping.freeOver ? 0 : shipping.fee;
  const discount = computeVoucherDiscount(voucher, subtotal, baseShipping);
  const isFreeShip = voucher?.type === 'freeShip' && subtotal >= voucher.minOrderValue;
  const shippingFee = isFreeShip ? 0 : baseShipping;
  const taxable = Math.max(subtotal - discount + shippingFee, 0);
  const tax = taxable * TAX_RATE;
  const totalAmount = Math.max(subtotal - discount + shippingFee + tax, 0);

  return {
    id: seed.id,
    userId: seed.userId,
    customerName: address.fullName,
    phoneNumber: address.phone,
    shippingAddress: address.address,
    items,
    subtotal,
    shippingFee,
    tax,
    discount,
    totalAmount,
    status: seed.status,
    currentStep: seed.status === 'delivered' ? 3 : ['pending', 'processing', 'shipped', 'cancelled'].indexOf(seed.status),
    canCancel: seed.status === 'pending' || seed.status === 'processing',
    cancelReason: seed.cancelReason,
    cancelledAt: seed.status === 'cancelled' ? seed.createdAt : undefined,
    createdAt: seed.createdAt,
    voucherCode: seed.voucherCode,
    paymentMethodId: payment.id,
    paymentMethodName: payment.name,
    shippingMethodId: shipping.id,
    shippingMethodName: shipping.name,
    trackingCode: seed.trackingCode ?? (seed.status === 'shipped' || seed.status === 'delivered'
      ? `VN${seed.id.replace('ORD-', '')}00`
      : undefined),
    timeline: buildTimeline(seed.status, seed.createdAt),
    note: seed.note,
  };
}

/* Bộ giả lập HTTP chung cho toàn bộ mock services. */

/** Bật/tắt độ trễ giả. Khi gọi API thật, set MOCK_LATENCY = false. */
export const MOCK_LATENCY = true;

/** Độ trễ ngẫu nhiên trong khoảng 150 - 600ms. */
export function mockDelay(): Promise<void> {
  if (!MOCK_LATENCY) return Promise.resolve();
  const ms = 150 + Math.floor(Math.random() * 450);
  return new Promise((resolve) => setTimeout(resolve, ms));
}

/** Lỗi chuẩn hoá như API thật trả về: { code, message }. */
export class ApiError extends Error {
  code: string;

  constructor(code: string, message: string) {
    super(message);
    this.name = 'ApiError';
    this.code = code;
  }
}

/** Ném ApiError ngay trong async function -> Promise bị reject. */
export function apiError(code: string, message: string): never {
  throw new ApiError(code, message);
}

/** Giả lập thành công: chờ delay rồi trả data. */
export async function respond<T>(data: T): Promise<T> {
  await mockDelay();
  return data;

import {

  Product,
  Review,
} from '../types';
import { apiError, respond } from './http';
import { MOCK_CATEGORIES } from '../mock/categories';
import { MOCK_BRANDS } from '../mock/brands';
import { MOCK_PRODUCTS } from '../mock/products';
import { MOCK_REVIEWS } from '../mock/reviews';

export interface Paged<T> {
  items: T[];
  page: number;
  limit: number;
  total: number;
  hasMore: boolean;
}

export type ProductSort = 'popular' | 'new' | 'priceAsc' | 'priceDesc' | 'rating';

export interface ProductQuery {
  q?: string;
  category?: string;
  categories?: string[];
  brands?: string[];
  minPrice?: number | null;
  maxPrice?: number | null;
  minRating?: number | null;
  sort?: ProductSort;
  page?: number;
  limit?: number;
}

/**
 * Giả lập Catalog API.
 *
 * GET /categories
 * GET /brands
 * GET /products?category=&q=&brand=&minPrice=&maxPrice=&minRating=&sort=&page=&limit=
 * GET /products/:id
 * GET /products/:id/reviews
 */
export const CatalogService = {
  /** GET /categories */
  async getCategories(): Promise<Category[]> {
    return respond(MOCK_CATEGORIES);
  },

  /** GET /brands */
  async getBrands(): Promise<Brand[]> {
    return respond(MOCK_BRANDS);
  },

  /** GET /products — phân trang + lọc + sắp xếp */
  async getProducts(query: ProductQuery = {}): Promise<Paged<Product>> {
    await respond(null);

    const page = query.page ?? 1;
    const limit = query.limit ?? 10;
    const q = (query.q ?? '').trim().toLowerCase();

    let result = MOCK_PRODUCTS.filter((p) => {
      if (q) {
        const haystack = `${p.name} ${p.brandName ?? ''} ${(p.tags ?? []).join(' ')}`.toLowerCase();
        if (!haystack.includes(q)) return false;
      }
      if (query.category && p.category !== query.category) return false;
      if (query.categories && query.categories.length > 0 && !query.categories.includes(p.category)) return false;
      if (query.brands && query.brands.length > 0 && !query.brands.includes(p.brandId ?? '')) return false;
      if (query.minPrice != null && p.price < query.minPrice) return false;
      if (query.maxPrice != null && p.price > query.maxPrice) return false;
      if (query.minRating != null && (p.rating ?? 0) < query.minRating) return false;
      return true;
    });

    switch (query.sort ?? 'popular') {
      case 'priceAsc':
        result = [...result].sort((a, b) => a.price - b.price);
        break;
      case 'priceDesc':
        result = [...result].sort((a, b) => b.price - a.price);
        break;
      case 'rating':
        result = [...result].sort((a, b) => (b.rating ?? 0) - (a.rating ?? 0));
        break;
      case 'new':
        result = [...result].sort((a, b) => (b.createdAt ?? '').localeCompare(a.createdAt ?? ''));
        break;
      default:
        result = [...result].sort((a, b) => (b.soldCount ?? 0) - (a.soldCount ?? 0));
    }

    const total = result.length;
    const start = (page - 1) * limit;
    const items = result.slice(start, start + limit);
    return { items, page, limit, total, hasMore: start + limit < total };
  },

  /** GET /products/:id */
  async getProductById(id: string): Promise<Product> {
    await respond(null);
    const product = MOCK_PRODUCTS.find((p) => p.id === id);
    if (!product) apiError('PRODUCT_NOT_FOUND', `Không tìm thấy sản phẩm #${id}.`);

  async getProductReviews(productId: string): Promise<Review[]> {
    await respond(null);
    return MOCK_REVIEWS.filter((r) => r.productId === productId);
  },

  /** GET /vouchers */
  async listVouchers(): Promise<Voucher[]> {

  },

  /**
   * POST /vouchers/validate
   * Trả về voucher hợp lệ hoặc ném ApiError với code VOUCHER_*.
   */
  async validateVoucher(
    code: string,
    subtotal: number,
    shippingFee = 0,
  ): Promise<{ voucher: Voucher; discount: number }> {
    await respond(null);
    const normalized = code.trim().toUpperCase();
    const voucher = MOCK_VOUCHERS.find((v) => v.code === normalized);

    if (!voucher) {
      apiError('VOUCHER_NOT_FOUND', `Mã ${normalized} không tồn tại.`);
    }
    if (!voucher.isActive) {
      apiError('VOUCHER_EXPIRED', `Mã ${normalized} đã hết hạn.`);
    }
    if (voucher.usedCount >= voucher.usageLimit) {
      apiError('VOUCHER_OUT_OF_USAGE', `Mã ${normalized} đã dùng hết lượt.`);
    }
    if (subtotal < voucher.minOrderValue) {
      apiError(
        'VOUCHER_MIN_ORDER_NOT_MET',
        `Không thể áp dụng mã ${normalized} cho tổng tiền dưới $${voucher.minOrderValue.toFixed(0)}.`,
      );
    }

    return {
      voucher: voucher as Voucher,
      discount: computeVoucherDiscount(voucher as Voucher, subtotal, shippingFee),
    };
  },

    const address = MOCK_ADDRESSES.find((a) => a.id === input.addressId) ?? MOCK_ADDRESSES[0];

    const isFreeShip = voucher?.type === 'freeShip' && subtotal >= voucher.minOrderValue;
    const shippingFee = isFreeShip ? 0 : baseShipping;
    const taxable = Math.max(subtotal - discount + shippingFee, 0);

      canCancel: true,
      createdAt: now,

      paymentMethodId: payment.id,

import { useCallback, useEffect, useState } from 'react';
import { ApiError } from '../services/http';

interface UseAsyncResult<T> {
  data: T | null;
  loading: boolean;
  error: ApiError | null;
  run: () => Promise<void>;
}

/** Chạy một hàm async, quản lý loading + error đơn giản. */
export function useAsync<T>(fn: () => Promise<T>, deps: readonly unknown[] = []): UseAsyncResult<T> {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<ApiError | null>(null);

  const run = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const result = await fn();
      setData(result);
    } catch (e) {
      setError(e instanceof ApiError ? e : new ApiError('UNKNOWN_ERROR', 'Đã có lỗi xảy ra.'));
    } finally {
      setLoading(false);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);

  useEffect(() => {
    void run();
  }, [run]);

  return { data, loading, error, run };
}

import { useEffect, useState } from 'react';

/** Trả về giá trị đã debounce sau delayMs ms. */
export function useDebounce<T>(value: T, delayMs = 300): T {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delayMs);
    return () => clearTimeout(timer);
  }, [value, delayMs]);

  return debounced;
}

export function CartProvider({
  children,
  resetKey = 0,
}: { children: ReactNode; resetKey?: number }) {
  const [items, setItems] = useState<CartItem[]>(seedCart);
  const [voucher, setVoucher] = useState<Voucher | null>(null);

  // Khi logout, AuthProvider bump resetKey -> reset giỏ.
  useEffect(() => {
    setItems(seedCart());
    setVoucher(null);
  }, [resetKey]);

  const addItem = useCallback((product: Product, quantity = 1, variantId?: string) => {
    const key = cartItemKey(product.id, variantId);
    setItems((prev) => {
      const existing = prev.find((i) => i.id === key);
      if (existing) {
        return prev.map((i) =>
          i.id === key ? { ...i, quantity: Math.min(10, i.quantity + quantity), selected: true } : i,
        );
      }
      const variant = product.variants?.find((v) => v.id === variantId);
      return [...prev, toCartItem(product, quantity, variant)];
    });
  }, []);

  const removeItem = useCallback((itemId: string) => {
    setItems((prev) => prev.filter((i) => i.id !== itemId));
  }, []);

  const updateQuantity = useCallback((itemId: string, delta: number) => {
    setItems((prev) =>
      prev.map((i) =>
        i.id === itemId
          ? { ...i, quantity: Math.max(1, Math.min(10, i.quantity + delta)) }
          : i,
      ),
    );
  }, []);

  const toggleSelect = useCallback((itemId: string) => {
    setItems((prev) =>
      prev.map((i) => (i.id === itemId ? { ...i, selected: !(i.selected ?? true) } : i)),
    );
  }, []);

  const applyVoucher = useCallback(async (code: string) => {
    const result = await VoucherService.validateVoucher(code, computeSubtotal(items));
    setVoucher(result.voucher);
    return result;
  }, [items]);

  const removeVoucher = useCallback(() => setVoucher(null), []);

  const clearSelected = useCallback(() => {
    setItems((prev) => prev.filter((i) => !i.selected));
  }, []);

  const clearCart = useCallback(() => {
    setItems([]);
    setVoucher(null);
  }, []);

  const reset = useCallback(() => {
    setItems(seedCart());
    setVoucher(null);
  }, []);

  const totalItems = useMemo(() => items.reduce((sum, i) => sum + i.quantity, 0), [items]);
  const selectedCount = useMemo(
    () => items.filter((i) => i.selected).reduce((sum, i) => sum + i.quantity, 0),
    [items],
  );

  const totals = useMemo(
    () => computeCartTotals(items, voucher, MOCK_SHIPPING_METHODS[0]),
    [items, voucher],
  );

  const value = useMemo(
    () => ({
      items,
      voucher,
      totalItems,
      selectedCount,
      totals,
      addItem,
      removeItem,
      updateQuantity,
      toggleSelect,
      applyVoucher,
      removeVoucher,
      clearSelected,
      clearCart,
      reset,
    }),
    [
      items, voucher, totalItems, selectedCount, totals,
      addItem, removeItem, updateQuantity, toggleSelect,
      applyVoucher, removeVoucher, clearSelected, clearCart, reset,
    ],
  );

  return <CartContext.Provider value={value}>{children}</CartContext.Provider>;
}

  useState,

} from 'react';
import { Order } from '../types';
import { getErrorMessage, ApiError } from '../services/http';
import { OrderService } from '../services/order.service';
import { useAuth } from './AuthContext';
import { useCart } from './CartContext';
import { MOCK_PRODUCTS } from '../mock/products';

interface OrdersContextValue {
  orders: Order[];
  loading: boolean;
  error: string | null;
  loadOrders: () => Promise<void>;
  getOrderById: (id: string) => Order | undefined;
  addOrder: (order: Order) => void;
  cancelOrder: (id: string, reason: string) => Promise<Order>;
  reorder: (id: string) => Promise<void>;
  reset: () => void;
}

const OrdersContext = createContext<OrdersContextValue | null>(null);

export function OrdersProvider({
  children,
  resetKey = 0,
}: { children: ReactNode; resetKey?: number }) {
  const { user } = useAuth();
  const { addItem } = useCart();
  const [orders, setOrders] = useState<Order[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const loadOrders = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const uid = user?.uid ?? 'user-demo';
      const data = await OrderService.listOrders(uid);
      setOrders(data);
    } catch (e) {
      setError(getErrorMessage(e));
    } finally {
      setLoading(false);
    }
  }, [user]);

  useEffect(() => {
    void loadOrders();
  }, [loadOrders, resetKey]);

  const getOrderById = useCallback(
    (id: string) => orders.find((o) => o.id === id),
    [orders],
  );

  const addOrder = useCallback((order: Order) => {
    setOrders((prev) => [order, ...prev]);
  }, []);

  const cancelOrder = useCallback(async (id: string, reason: string) => {
    const updated = await OrderService.cancelOrder(id, reason);
    setOrders((prev) => prev.map((o) => (o.id === id ? updated : o)));
    return updated;
  }, []);

  const reorder = useCallback(async (id: string) => {
    const items = await OrderService.reorder(id);
    items.forEach((item) => {
      const product = MOCK_PRODUCTS.find((p) => p.id === item.productId);
      if (product) {
        addItem(product, item.quantity);
      }
    });
  }, [addItem]);

  const reset = useCallback(() => {
    void loadOrders();
  }, [loadOrders]);

  const value = useMemo(
    () => ({
      orders,
      loading,
      error: error as string | null,
      loadOrders,
      getOrderById,
      addOrder,
      cancelOrder,
      reorder,
      reset,
    }),
    [orders, loading, error, loadOrders, getOrderById, addOrder, cancelOrder, reorder, reset],
  );

  return <OrdersContext.Provider value={value}>{children}</OrdersContext.Provider>;
}

export function useOrders(): OrdersContextValue {
  const ctx = useContext(OrdersContext);
  if (!ctx) {
    throw new Error('useOrders phải được dùng bên trong <OrdersProvider>');
  }

export function CheckoutProvider({
  children,
  resetKey = 0,

  const { user } = useAuth();
  const cart = useCart();
  const orders = useOrders();
  const notifications = useNotifications();
  const addresses = useAddresses();
  const payments = usePaymentMethods();

  const [addressId, setAddressId] = useState<string | null>(null);
  const [paymentMethodId, setPaymentMethodId] = useState<string | null>(null);
  const [shippingMethodId, setShippingMethodId] = useState<string | null>(MOCK_SHIPPING_METHODS[0].id);
  const [note, setNote] = useState('');
  const [lastPlacedOrder, setLastPlacedOrder] = useState<Order | null>(null);
  const [placing, setPlacing] = useState(false);

  // Tự chọn địa chỉ + thẻ mặc định khi dữ liệu tải xong.
  useEffect(() => {
    if (!addressId && addresses.defaultAddress) {
      setAddressId(addresses.defaultAddress.id);
    }
  }, [addressId, addresses.defaultAddress]);

  useEffect(() => {
    if (!paymentMethodId && payments.defaultMethod) {
      setPaymentMethodId(payments.defaultMethod.id);
    }
  }, [paymentMethodId, payments.defaultMethod]);

  useEffect(() => {
    setAddressId(null);
    setPaymentMethodId(null);
    setNote('');
    setLastPlacedOrder(null);
  }, [resetKey]);

  const placeOrder = useCallback(async (): Promise<Order> => {
    if (!user) {
      throw new ApiError('AUTH_REQUIRED', 'Vui lòng đăng nhập trước khi thanh toán.');
    }

    const selectedItems = cart.items.filter((item) => item.selected);
    if (selectedItems.length === 0) {
      throw new ApiError('CART_EMPTY', 'Giỏ hàng trống hoặc chưa chọn sản phẩm.');
    }

    const address = addresses.addresses.find((a) => a.id === addressId) ?? addresses.defaultAddress;
    if (!address) {
      throw new ApiError('ADDRESS_REQUIRED', 'Vui lòng chọn địa chỉ giao hàng.');
    }

    const payment = payments.methods.find((p) => p.id === paymentMethodId) ?? payments.defaultMethod;
    if (!payment) {
      throw new ApiError('PAYMENT_REQUIRED', 'Vui lòng chọn phương thức thanh toán.');
    }

    const shipping =
      MOCK_SHIPPING_METHODS.find((s) => s.id === shippingMethodId) ?? MOCK_SHIPPING_METHODS[0];
    const totals = computeCartTotals(cart.items, cart.voucher, shipping);

    setPlacing(true);
    try {
      const created = await OrderService.createOrder({
        userId: user.uid,
        customerName: address.fullName,
        phoneNumber: address.phone,
        shippingAddress: address.address,
        items: selectedItems.map((item) => ({

          name: item.product.name,
          price: item.product.price,
          quantity: item.quantity,
          imageUrl: item.product.imageUrl,
          variantId: item.variantId,
          variantName: item.variantName,
        })),
        addressId: address.id,
        paymentMethodId: payment.id,
        paymentMethodName: payment.name,
        shippingMethodId: shipping.id,
        shippingMethodName: shipping.name,
        voucherCode: cart.voucher?.code,
        note: note.trim() || undefined,
      });

      orders.addOrder(created);
      cart.clearSelected();
      cart.removeVoucher();
      setLastPlacedOrder(created);

      await notifications.addNotification({
        type: 'order',
        title: 'Đặt hàng thành công',
        message: `Đơn ${created.id} đã được ghi nhận. Tổng ${totals.total.toFixed(2)}$ (đã gồm thuế).`,
        data: { orderId: created.id },
      });

      return created;
    } finally {
      setPlacing(false);
    }
  }, [user, cart, orders, notifications, addresses, payments, addressId, paymentMethodId, shippingMethodId, note]);

  const reset = useCallback(() => {
    setAddressId(null);
    setPaymentMethodId(null);
    setShippingMethodId(MOCK_SHIPPING_METHODS[0].id);
    setNote('');
    setLastPlacedOrder(null);
  }, []);

  const value = useMemo(
    () => ({
      addressId,
      paymentMethodId,
      shippingMethodId,
      note,
      lastPlacedOrder,
      placing,
      setAddressId,
      setPaymentMethodId,
      setShippingMethodId,
      setNote,
      placeOrder,
      reset,
    }),
    [addressId, paymentMethodId, shippingMethodId, note, lastPlacedOrder, placing, placeOrder, reset],
  );

  return <CheckoutContext.Provider value={value}>{children}</CheckoutContext.Provider>;
}

interface ProductFilterContextValue {

  setQuery: (query: string) => void;
  setCategoryIds: (ids: string[]) => void;
  setBrandIds: (ids: string[]) => void;
  setPriceRange: (min: number | null, max: number | null) => void;
  setMinRating: (rating: number | null) => void;
  setSort: (sort: ProductSort) => void;
  /** Áp nguyên bộ lọc từ màn Filter. */
  applyFilters: (patch: Partial<ProductFilterState>) => void;
  resetFilters: () => void;
  isFiltered: boolean;
}

const INITIAL_FILTERS: ProductFilterState = {
  query: '',
  categoryIds: [],
  brandIds: [],
  minPrice: null,
  maxPrice: null,
  minRating: null,
  sort: 'popular',
};

const ProductFilterContext = createContext<ProductFilterContextValue | null>(null);

export function ProductFilterProvider({ children }: { children: ReactNode }) {
  const [filters, setFilters] = useState<ProductFilterState>(INITIAL_FILTERS);

  const setQuery = useCallback((query: string) => {
    setFilters((prev) => ({ ...prev, query }));
  }, []);

  const setCategoryIds = useCallback((ids: string[]) => {
    setFilters((prev) => ({ ...prev, categoryIds: ids }));
  }, []);

  const setBrandIds = useCallback((ids: string[]) => {
    setFilters((prev) => ({ ...prev, brandIds: ids }));
  }, []);

  const setPriceRange = useCallback((min: number | null, max: number | null) => {
    setFilters((prev) => ({ ...prev, minPrice: min, maxPrice: max }));
  }, []);

  const setMinRating = useCallback((rating: number | null) => {
    setFilters((prev) => ({ ...prev, minRating: rating }));
  }, []);

  const setSort = useCallback((sort: ProductSort) => {
    setFilters((prev) => ({ ...prev, sort }));
  }, []);

  const applyFilters = useCallback((patch: Partial<ProductFilterState>) => {
    setFilters((prev) => ({ ...prev, ...patch }));
  }, []);

  const resetFilters = useCallback(() => {
    setFilters(INITIAL_FILTERS);
  }, []);

  const isFiltered = useMemo(
    () =>
      filters.query.trim().length > 0 ||
      filters.categoryIds.length > 0 ||
      filters.brandIds.length > 0 ||
      filters.minPrice != null ||
      filters.maxPrice != null ||
      filters.minRating != null,
    [filters],
  );

  const value = useMemo(
    () => ({
      filters,
      setQuery,
      setCategoryIds,
      setBrandIds,
      setPriceRange,
      setMinRating,
      setSort,
      applyFilters,
      resetFilters,
      isFiltered,
    }),
    [
      filters, setQuery, setCategoryIds, setBrandIds, setPriceRange,
      setMinRating, setSort, applyFilters, resetFilters, isFiltered,
    ],
  );

import { Pressable, StyleSheet, Text, View } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { colors, radius } from '../constants/theme';

interface QuantityStepperProps {
  value: number;
  onChange: (value: number) => void;
  min?: number;
  max?: number;
}

/** Nút tăng/giảm số lượng, chặn đúng min/max. */
export function QuantityStepper({ value, onChange, min = 1, max = 10 }: QuantityStepperProps) {
  const decrease = () => onChange(Math.max(min, value - 1));
  const increase = () => onChange(Math.min(max, value + 1));

  return (
    <View style={styles.container}>
      <Pressable
        onPress={decrease}
        disabled={value <= min}
        style={({ pressed }) => [styles.button, pressed && styles.pressed]}
        accessibilityRole="button"
        accessibilityLabel="Giảm số lượng"
      >
        <Ionicons name="remove" size={16} color={value <= min ? colors.grayLight : colors.dark} />
      </Pressable>

      <Text style={styles.value}>{value}</Text>

      <Pressable
        onPress={increase}
        disabled={value >= max}
        style={({ pressed }) => [styles.button, pressed && styles.pressed]}
        accessibilityRole="button"
        accessibilityLabel="Tăng số lượng"
      >
        <Ionicons name="add" size={16} color={value >= max ? colors.grayLight : colors.dark} />
      </Pressable>
    </View>
  );
}

import React, { useEffect, useRef } from 'react';
import { Animated, StyleSheet, Text } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { colors, radius, spacing } from '../constants/theme';

interface ToastProps {
  message: string;
  visible: boolean;
  onHide: () => void;
  duration?: number;
}

/** Toast nhỏ gọn (Animated, không cần thư viện ngoài). */
export function Toast({ message, visible, onHide, duration = 2000 }: ToastProps) {
  const opacity = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    if (!visible) {
      opacity.setValue(0);
      return;
    }
    Animated.timing(opacity, { toValue: 1, duration: 200, useNativeDriver: true }).start();
    const timer = setTimeout(() => {
      Animated.timing(opacity, { toValue: 0, duration: 200, useNativeDriver: true }).start(() => onHide());
    }, duration);
    return () => clearTimeout(timer);
  }, [visible, duration, onHide, opacity]);

  if (!visible) return null;

  return (
    <Animated.View style={[styles.toast, { opacity }]} accessibilityLiveRegion="polite">
      <Ionicons name="checkmark-circle" size={20} color="#4ADE80" />
      <Text style={styles.text}>{message}</Text>
    </Animated.View>
  );
}

const styles = StyleSheet.create({

    position: 'absolute',
    left: 24,

/** Điều hướng chính + route guard. */
function RootNavigator() {
  const { user, isLoading } = useAuth();
  const segments = useSegments();
  const router = useRouter();

  useEffect(() => {
    if (isLoading) return;

    const isInAuthGroup = segments[0] === '(auth)';

    if (!user && !isInAuthGroup) {
      // Chưa đăng nhập -> buộc về màn login.
      router.replace('/(auth)/login');
    } else if (user && isInAuthGroup) {
      // Đã đăng nhập mà đang ở màn auth -> về Home.
      router.replace('/(tabs)/home');
    }
  }, [user, isLoading, segments, router]);

  return (
    <>
      <Stack screenOptions={{ headerShown: false, animation: 'slide_from_right' }}>
        <Stack.Screen name="(tabs)" options={{ headerShown: false }} />
        <Stack.Screen name="(auth)" options={{ headerShown: false }} />
        <Stack.Screen name="product/[id]" options={{ headerShown: false }} />
        <Stack.Screen name="cart" options={{ headerShown: false }} />
        <Stack.Screen name="checkout" options={{ headerShown: false }} />
        <Stack.Screen name="checkout-success" options={{ headerShown: false }} />
        <Stack.Screen name="orders" options={{ headerShown: false }} />
        <Stack.Screen name="order/[id]" options={{ headerShown: false }} />
        <Stack.Screen name="notification" options={{ headerShown: false }} />
        <Stack.Screen name="shipping-address" options={{ headerShown: false }} />
        <Stack.Screen name="payment-method" options={{ headerShown: false }} />
        <Stack.Screen name="help" options={{ headerShown: false }} />
        <Stack.Screen name="update-user" options={{ headerShown: false }} />
        <Stack.Screen name="app-settings" options={{ headerShown: false }} />

        {/* Chỉ filter là bottom sheet modal */}
        <Stack.Screen
          name="filter"
          options={{
            presentation: 'transparentModal',
            animation: 'slide_from_bottom',
            headerShown: false,
          }}
        />
      </Stack>

      {isLoading ? <SplashOverlay /> : null}
    </>
  );
}

export default function RootLayout() {
  return (
    <AppProviders>
      <StatusBar style="dark" />
      <RootNavigator />
    </AppProviders>
  );
}

## 5. Context và mối quan hệ

```
AuthProvider
└─ NotificationsProvider
   └─ CartProvider
      └─ FavouritesProvider
         └─ AddressesProvider
            └─ PaymentMethodsProvider
               └─ SettingsProvider
                  └─ OrdersProvider      (dùng useCart để mua lại)
                     └─ CheckoutProvider (gọi Cart + Orders + Notifications + Addresses + Payments)
                        └─ ProductFilterProvider
```

Luồng đặt hàng (`CheckoutContext.placeOrder`):
1. Kiểm tra đăng nhập, giỏ hàng, địa chỉ, phương thức thanh toán.
2. Gọi `OrderService.createOrder` → tạo Order thật.
3. Đưa order vào `OrdersContext`.
4. Xoá sản phẩm đã mua khỏi Cart, xoá voucher.
5. Sinh thông báo `order` trong `NotificationsContext`.

Khi logout, `AppProviders` tăng `resetKey` để Cart/Favourites/Addresses/... reset về trạng thái ban đầu.

## 6. Cách thay mock bằng API thật

1. Đặt `MOCK_LATENCY = false` trong `src/services/http.ts` (hoặc xoá bước delay).
2. Thay từng thân hàm trong `src/services` bằng `fetch`/`axios` gọi đúng endpoint đã liệt kê ở trên.
3. **Giữ nguyên chữ ký hàm** — UI và context đã phụ thuộc vào các tên hàm này.
4. Ép response thành các type trong `src/types/index.ts`.
5. Với mutation (đơn hàng, thông báo, địa chỉ...), bỏ biến `xxxDb` trong service, gọi API thật rồi return response.
6. Thay `AuthProvider` bootstrap bằng `GET /auth/me` để khôi phục phiên đăng nhập từ token.

> Kiểm tra lại sau khi đổi: chạy `npx tsc --noEmit` đảm bảo không lỗi type.
